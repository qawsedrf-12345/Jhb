local blurEffect = nil
local blurTween = nil

local function criarBlur(tamanho)
    if blurEffect and blurEffect.Parent then blurEffect:Destroy() end
    if blurTween then blurTween:Cancel() blurTween = nil end
    blurEffect = Instance.new("BlurEffect")
    blurEffect.Name = "RP_Blur"
    blurEffect.Size = 0
    blurEffect.Parent = Lighting
    blurTween = TweenService:Create(blurEffect, TweenInfo.new(0.1, Enum.EasingStyle.Quad), {Size = tamanho})
    blurTween:Play()
end

local function removerBlur(duracao)
    if not blurEffect then return end
    local b = blurEffect
    if blurTween then blurTween:Cancel() blurTween = nil end
    blurTween = TweenService:Create(b, TweenInfo.new(duracao or 0.4, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Size = 0})
    blurTween:Play()
    task.delay(duracao or 0.4, function()
        if b and b.Parent then b:Destroy() end
        if blurEffect == b then blurEffect = nil blurTween = nil end
    end)
end

local function cameraShake(intensidade, duracao)
    local camera = workspace.CurrentCamera
    if not camera then return end
    local inicio = tick()
    local conn
    conn = RunService.RenderStepped:Connect(function()
        if tick() - inicio >= duracao then
            conn:Disconnect()
            return
        end
        local p = 1 - ((tick() - inicio) / duracao)
        local sx = (math.random() - 0.5) * 2 * intensidade * p / 100
        local sy = (math.random() - 0.5) * 2 * intensidade * p / 100
        camera.CFrame = camera.CFrame * CFrame.new(sx, sy, 0)
    end)
end

local function fecharPortal(portalPart)
    if not portalPart or not portalPart.Parent then return end
    if portalPart:GetAttribute("Fechando") then return end
    portalPart:SetAttribute("Fechando", true)

    portalPart.CanTouch = false
    portalPart.CanCollide = false

    for _, emitter in ipairs(portalPart:GetChildren()) do
        if emitter:IsA("ParticleEmitter") then
            emitter.Enabled = false
        end
    end

    local somLoop = portalPart:FindFirstChild("PortalLoopSound")
    if somLoop and somLoop:IsA("Sound") then
        pcall(function()
            TweenService:Create(somLoop, TweenInfo.new(0.35, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Volume = 0}):Play()
        end)
    end

    for _, sfx in ipairs(portalPart:GetChildren()) do
        if sfx:IsA("Sound") and sfx.Name ~= "PortalLoopSound" then
            pcall(function()
                TweenService:Create(sfx, TweenInfo.new(0.3), {Volume = 0}):Play()
            end)
        end
    end

    local DUR = 0.45
    local rotacaoTotal = math.rad(720)
    local _, yRotInicial = portalPart.CFrame:ToOrientation()
    local tamanhoInicial = portalPart.Size

    local inicio = tick()
    local conn
    conn = RunService.Heartbeat:Connect(function()
        if not portalPart or not portalPart.Parent then
            if conn then conn:Disconnect() end
            return
        end

        local p = (tick() - inicio) / DUR
        if p >= 1 then
            conn:Disconnect()
            if portalPart and portalPart.Parent then
                portalPart:Destroy()
            end
            return
        end

        local novoY = yRotInicial + rotacaoTotal * p
        portalPart.CFrame = CFrame.new(portalPart.Position) * CFrame.Angles(0, novoY, 0)

        local fator = 1 - p
        portalPart.Size = Vector3.new(
            tamanhoInicial.X * fator,
            tamanhoInicial.Y * fator,
            math.max(tamanhoInicial.Z * fator, 0.01)
        )
    end)
end

local function wipePortalsPorTag(tag, comEfeito)
    local portais = {}
    local vistos = {}

    for _, p in ipairs(CollectionService:GetTagged(tag)) do
        if p and not vistos[p] then
            vistos[p] = true
            table.insert(portais, p)
        end
    end

    if comEfeito then
        for _, p in ipairs(portais) do
            if p and p.Parent then
                fecharPortal(p)
            end
        end
        if #portais > 0 then
            efeitoTP()
        end
    else
        for _, p in ipairs(portais) do
            if p and p.Parent then
                p:Destroy()
            end
        end
    end

    return #portais
end

local function wipeAllPortals(comEfeito)
    local total = wipePortalsPorTag(TAG_GEN, comEfeito)
    table.clear(activePortals)
    return total
end

local function deletarVoidPlace()
    voidForaDesde = nil
    if voidPlaceFolder then
        voidPlaceFolder:Destroy()
        voidPlaceFolder = nil
    end
    if voidMonitorConn then
        local conn = voidMonitorConn
        voidMonitorConn = nil
        conn:Disconnect()
    end
end

local function monitorarSaidaDoVoid()
    if voidMonitorConn then voidMonitorConn:Disconnect() end

    voidMonitorConn = RunService.Heartbeat:Connect(function()
        if not voidPlaceFolder then return end

        local char = player.Character
        if not char then
            voidForaDesde = nil
            return
        end

        local humanoid = char:FindFirstChildOfClass("Humanoid")
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end

        -- ⚠️ Se morrer, deleta na hora (sem esperar 3s)
        if humanoid and humanoid.Health <= 0 then
            voidForaDesde = nil
            task.defer(deletarVoidPlace)
            task.defer(deletarLocalTP)
            return
        end

        local pos = hrp.Position
        local foraX = math.abs(pos.X) > VOID_TAMANHO + 20
        local foraZ = math.abs(pos.Z) > VOID_TAMANHO + 20
        local caiuMuito = pos.Y < VOID_Y - 100
        local subiuMuito = pos.Y > VOID_Y + 500

        local estaFora = foraX or foraZ or caiuMuito or subiuMuito

        if estaFora then
            -- 🕐 Começou a contar agora?
            if not voidForaDesde then
                voidForaDesde = tick()
            end

            -- ⏱️ Já passou da tolerância?
            if tick() - voidForaDesde >= VOID_TOLERANCIA then
                voidForaDesde = nil
                task.defer(deletarVoidPlace)
                task.defer(deletarLocalTP)
            end
        else
            -- ✅ Voltou pra dentro — reseta o timer
            if voidForaDesde then
                voidForaDesde = nil
            end
        end
    end)
end

local function createSaggingCable(startX, startY, endX, endY, sagDepth, wireColor, customZIndex)
    local z = customZIndex or 6
    local segments = 6
    for i = 0, segments - 1 do
        local t1 = i / segments
        local t2 = (i + 1) / segments
        local function getPoint(t)
            local x = startX + (endX - startX) * t
            local y = startY + (endY - startY) * t + sagDepth * (4 * t * (1 - t))
            return x, y
        end
        local x1, y1 = getPoint(t1)
        local x2, y2 = getPoint(t2)
        local dx = x2 - x1
        local dy = y2 - y1
        local dist = math.sqrt(dx * dx + dy * dy)
        local angle = math.deg(math.atan(dy, dx))

        local seg = Instance.new("Frame")
        seg.Size = UDim2.new(0, dist + 1, 0, 2)
        seg.Position = UDim2.new(0, x1, 0, y1)
        seg.Rotation = angle
        seg.BackgroundColor3 = wireColor
        seg.BorderSizePixel = 0
        seg.ZIndex = z
        seg.Parent = mainFrame

        if i == math.floor(segments / 2) then
            local spark = Instance.new("Frame")
            spark.Size = UDim2.new(0, 3, 0, 3)
            spark.Position = UDim2.new(0.5, -1.5, 0.5, -1.5)
            spark.BackgroundColor3 = Color3.fromRGB(255, 140, 0)
            spark.BorderSizePixel = 0
            spark.ZIndex = z + 2
            spark.Parent = seg
            local sc = Instance.new("UICorner")
            sc.CornerRadius = UDim.new(1, 0)
            sc.Parent = spark
            task.spawn(function()
                while spark and spark.Parent do
                    spark.Visible = math.random() > 0.3
                    spark.BackgroundColor3 = math.random() > 0.5
                        and Color3.fromRGB(255, 100, 0)
                        or Color3.fromRGB(255, 220, 50)
                    task.wait(math.random(10, 25) / 100)
                end
            end)
        end
    end
end

local function createPortalInstance(cframe, tag, opcoes)
    opcoes = opcoes or {}
    tag = tag or TAG_GEN
    local pos = cframe.Position
    local _, yRot, _ = cframe:ToOrientation()
    local verticalCFrame = CFrame.new(pos) * CFrame.Angles(0, yRot, 0)

    local portalPart = Instance.new("Part")
    portalPart.Size = Vector3.new(0.1, 0.1, 0.1)
    portalPart.CFrame = verticalCFrame
    portalPart.Anchored = true
    portalPart.CanCollide = false
    portalPart.Transparency = 1
    portalPart.Parent = workspace
    CollectionService:AddTag(portalPart, tag)
    if tag == TAG_GEN then table.insert(activePortals, portalPart) end

    if opcoes.comSom ~= false then
        local portalLoop = Instance.new("Sound")
        portalLoop.Name = "PortalLoopSound"
        portalLoop.SoundId = PORTAL_LOOP_SOUND_ID
        portalLoop.Volume = 2
        portalLoop.Looped = true
        portalLoop.RollOffMaxDistance = 150
        portalLoop.RollOffMode = Enum.RollOffMode.InverseTapered
        portalLoop.Parent = portalPart
        portalLoop:Play()
    end

    local portalLight = Instance.new("PointLight")
    portalLight.Color = Color3.fromRGB(50, 255, 80)
    portalLight.Range = 15
    portalLight.Brightness = 3
    portalLight.Parent = portalPart

    if opcoes.comParticulas ~= false then
        local sparks = Instance.new("ParticleEmitter")
        sparks.Color = ColorSequence.new(Color3.fromRGB(50, 255, 80))
        sparks.Size = NumberSequence.new({
            NumberSequenceKeypoint.new(0, 0.3),
            NumberSequenceKeypoint.new(1, 0)
        })
        sparks.Lifetime = NumberRange.new(0.5, 1.2)
        sparks.Rate = 25
        sparks.Speed = NumberRange.new(3, 7)
        sparks.SpreadAngle = Vector2.new(360, 360)
        sparks.EmissionDirection = Enum.NormalId.Front
        sparks.Parent = portalPart
    end

    local function createSurfaceFace(normalFace)
        local surf = Instance.new("SurfaceGui")
        surf.Face = normalFace
        surf.SizingMode = Enum.SurfaceGuiSizingMode.PixelsPerStud
        surf.PixelsPerStud = 50
        surf.LightInfluence = 0
        surf.Parent = portalPart

        local img = Instance.new("ImageLabel")
        img.Size = UDim2.new(1, 0, 1, 0)
        img.BackgroundTransparency = 1
        img.Image = PORTAL_TEXTURE
        img.AnchorPoint = Vector2.new(0.5, 0.5)
        img.Position = UDim2.new(0.5, 0, 0.5, 0)
        img.Parent = surf

        local c = Instance.new("UICorner")
        c.CornerRadius = UDim.new(0.5, 0)
        c.Parent = img
        return img
    end

    local fFront = createSurfaceFace(Enum.NormalId.Front)
    local bFront = createSurfaceFace(Enum.NormalId.Back)

    TweenService:Create(
        portalPart,
        TweenInfo.new(0.45, Enum.EasingStyle.Back, Enum.EasingDirection.Out),
        { Size = Vector3.new(7, 7, 0.1) }
    ):Play()

    if opcoes.comGiro ~= false then
        task.spawn(function()
            while portalPart and portalPart.Parent do
                local fRot = (fFront.Rotation + 8) % 360
                fFront.Rotation = fRot
                bFront.Rotation = fRot
                task.wait(0.03)
            end
        end)
    end

    return portalPart, fFront, bFront
end

local function spawnPortalSaida(char)
    if not char or not char.Parent then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end

    local cframeInicial = root.CFrame * CFrame.new(0, 0, -2)

    local portal, fFront, bFront = createPortalInstance(cframeInicial, TAG_SAIDA, {
        comSom = false,
        comParticulas = false,
        comGiro = false,
    })

    portal.Transparency = 1

    if fFront then fFront.ImageTransparency = 0 end
    if bFront then bFront.ImageTransparency = 0 end

    local cframeFinal = cframeInicial * CFrame.new(0, 0, 4)

    task.spawn(function()
        local DUR = 0.6
        local inicio = tick()

        while portal and portal.Parent do
            local p = (tick() - inicio) / DUR
            if p >= 1 then break end

            local posAtual = cframeInicial:Lerp(cframeFinal, p)
            portal.CFrame = CFrame.new(posAtual.Position) * (cframeInicial - cframeInicial.Position)

            local transparencia = math.clamp(p, 0, 1)
            if fFront then fFront.ImageTransparency = transparencia end
            if bFront then bFront.ImageTransparency = transparencia end

            task.wait()
        end

        if portal and portal.Parent then
            portal:Destroy()
        end
    end)

    return portal
end

local function limparMarcacao()
    for _, conn in ipairs(marcacaoConns) do
        if conn and conn.Disconnect then pcall(function() conn:Disconnect() end) end
    end
    table.clear(marcacaoConns)

    if marcacaoP1 and marcacaoP1.Parent then fecharPortal(marcacaoP1) end
    if marcacaoP2 and marcacaoP2.Parent then fecharPortal(marcacaoP2) end
    marcacaoP1, marcacaoP2 = nil, nil

    marcacaoBtn.Text = "🎯 Marcação [0/2]"
    marcacaoBtn.BackgroundColor3 = Color3.fromRGB(255, 200, 50)
    marcacaoBtn.TextColor3 = Color3.fromRGB(50, 30, 0)
    if marcacaoStroke then marcacaoStroke.Color = Color3.fromRGB(180, 130, 0) end
    marcacaoBtn:SetAttribute("ActiveState", 0)
    getgenv().Marcacao_Ativo = false
end

local function conectarPortais(p1, p2)
    local isTeleporting = false
    local COOLDOWN = 1.0

    local function irPara(hit, destino)
        local char = player.Character
        if isTeleporting or not char or hit.Parent ~= char then return end
        if not destino or not destino.Parent then return end
        isTeleporting = true

        local root = char:FindFirstChild("HumanoidRootPart")
        if root then
            fazerTPComEfeito(root, CFrame.new(destino.Position + Vector3.new(0, 3, 0)))
        end

        task.delay(0.5, function()
            limparMarcacao()
            task.delay(COOLDOWN, function()
                isTeleporting = false
            end)
        end)
    end

    table.insert(marcacaoConns, p1.Touched:Connect(function(h) irPara(h, p2) end))
    table.insert(marcacaoConns, p2.Touched:Connect(function(h) irPara(h, p1) end))
end

                local targetChar = p.Character
                if not targetChar then
                    tpPlayerBtn.Text = "No Target!"
                    task.wait(1.5)
                    tpPlayerBtn.Text = "Select Player"
                    return
                end

                local targetPivot = targetChar:GetPivot()
                local actualRoot = targetChar:FindFirstChild("HumanoidRootPart") or targetChar.PrimaryPart

                local currentChar = player.Character
                local currentRoot = currentChar and currentChar:FindFirstChild("HumanoidRootPart")

                    currentRoot.CFrame = actualRoot.CFrame * CFrame.new(0, 0, -4)

                else
                    currentRoot.CFrame = targetPivot * CFrame.new(0, 5, 0)
                    currentRoot.Anchored = true
                    pcall(function()
                        player:RequestStreamAroundAsync(targetPivot.Position)
                    end)

                    local timeout = tick() + 5
                    while tick() < timeout do
                        actualRoot = targetChar:FindFirstChild("HumanoidRootPart") or targetChar.PrimaryPart
                        if actualRoot then break end
                        task.wait(0.05)
                    end

                        currentRoot.CFrame = actualRoot.CFrame * CFrame.new(0, 0, -4)

                    end
                    currentRoot.Anchored = false
                    tpPlayerBtn.Text = "Select Player"
                end

                task.wait(0.05)
                local currentChar2 = player.Character
                if currentChar2 then
                    spawnPortalSaida(currentChar2)
                end
            end)
        end)
    end)

    return pBtn
end

-- Função de refresh incremental
local function refreshPlayerList()
    -- ✅ Remove botões de quem saiu
    for userId, btn in pairs(playerBtns) do
        local jogador = Players:GetPlayerByUserId(userId)
        if not jogador or jogador == player then
            if btn and btn.Parent then btn:Destroy() end
            playerBtns[userId] = nil
        end
    end

    -- ✅ Adiciona botões de quem entrou
    local count = 0
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player then
            count = count + 1

            if playerBtns[p.UserId] then
                -- Já existe — só atualiza LayoutOrder
                playerBtns[p.UserId].LayoutOrder = count
            else
                -- Cria novo
                playerBtns[p.UserId] = criarBotaoPlayer(p, count)
            end
        end
    end

    scrollingList.CanvasSize = UDim2.new(0, 0, 0, count * 19)
    return count
end

local function antiTP_bloquear(posicaoAnterior, character)
    local hrp = character:FindFirstChild("HumanoidRootPart")
    if not hrp then return end

    local rotacaoAtual = hrp.CFrame - hrp.CFrame.Position
    hrp.CFrame = CFrame.new(posicaoAnterior + Vector3.new(0, ATP.OffsetY, 0)) * rotacaoAtual
    hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
    hrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)

    alertaVermelho()
    efeitoTP()
end

RunService.Heartbeat:Connect(function()
    if not getgenv().AntiTP_Ativo then return end

    local character = player.Character
    if not character then
        atpUltPos = nil
        return
    end

    local hrp = character:FindFirstChild("HumanoidRootPart")
    local humanoid = character:FindFirstChildOfClass("Humanoid")

    if not hrp or not humanoid then
        atpUltPos = nil
        return
    end

    if humanoid.Health <= 0 then
        atpUltPos = nil
        return
    end

    if atpCooldown then return end

    local agora = os.clock()

    if agora - atpUltimoPulo < ATP.JanelaPulo then
        atpUltPos = hrp.Position
        atpUltimoCheck = agora
        return
    end

    local state = humanoid:GetState()
    if state == Enum.HumanoidStateType.Jumping
        or state == Enum.HumanoidStateType.Freefall
    then
        atpUltPos = hrp.Position
        atpUltimoCheck = agora
        return
    end

    if agora - atpUltimoCheck < ATP.Cooldown then return end

    if not atpUltPos then
        atpUltPos = hrp.Position
        atpUltimoCheck = agora
        return
    end

    if agora - atpUltimoRespawn < ATP.IgnorarJanela then
        atpUltPos = hrp.Position
        atpUltimoCheck = agora
        return
    end

    if ATP.IgnorarSeCaindo then
        local velY = hrp.AssemblyLinearVelocity.Y
        if velY < ATP.VelocidadeCaindoMax then
            atpUltPos = hrp.Position
            atpUltimoCheck = agora
            return
        end
    end

    local distancia = (hrp.Position - atpUltPos).Magnitude

    if distancia > ATP.DistanciaMax then
        atpCooldown = true
        local posAnterior = atpUltPos
        antiTP_bloquear(posAnterior, character)

        atpUltPos = posAnterior + Vector3.new(0, ATP.OffsetY, 0)
        atpUltimoCheck = agora

        task.delay(0.3, function() atpCooldown = false end)
        return
    end

    atpUltPos = hrp.Position
    atpUltimoCheck = agora
end)

local function fullCleanup()
    getgenv().AntiTP_Ativo = false
    getgenv().Marcacao_Ativo = false

    wipePortalsPorTag(TAG_GEN, true)
    wipePortalsPorTag(TAG_M1, true)
    wipePortalsPorTag(TAG_M2, true)
    wipePortalsPorTag(TAG_VOID, true)
    wipePortalsPorTag(TAG_RET, true)
    wipePortalsPorTag(TAG_SAIDA, true)
    table.clear(activePortals)

    deletarVoidPlace()
    deletarLocalTP()
    limparMarcacao()

    if blurEffect and blurEffect.Parent then
        blurEffect:Destroy()
        blurEffect = nil
    end
    blurTween = nil

    for _, s in ipairs(SoundService:GetChildren()) do
        if s:IsA("Sound") then
            pcall(function() s:Destroy() end)
        end
    end

    if screenGui and screenGui.Parent then screenGui:Destroy() end
    if fxGui and fxGui.Parent then fxGui:Destroy() end

    getgenv().RickPortalCleanup = nil
end

getgenv().RickPortalCleanup = fullCleanup
