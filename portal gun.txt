local blurEffect = nil
local blurTween = nil

local function criarBlur(tamanho)
    if blurEffect and blurEffect.Parent then blurEffect:Destroy() end
    if blurTween then blurTween:Cancel() blurTween = nil end
    blurEffect = Instance.new("BlurEffect")
    blurEffect.Name = "RP_Blur"
    blurEffect.Size = 0
    blurEffect.Parent = Lighting
    blurTween = TweenService:Create(blurEffect, TweenInfo.new(0.1, Enum.EasingStyle.Quad), {Size = tamanho})
    blurTween:Play()
end

local function removerBlur(duracao)
    if not blurEffect then return end
    local b = blurEffect
    if blurTween then blurTween:Cancel() blurTween = nil end
    blurTween = TweenService:Create(b, TweenInfo.new(duracao or 0.4, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Size = 0})
    blurTween:Play()
    task.delay(duracao or 0.4, function()
        if b and b.Parent then b:Destroy() end
        if blurEffect == b then blurEffect = nil blurTween = nil end
    end)
end

local function cameraShake(intensidade, duracao)
    local camera = workspace.CurrentCamera
    if not camera then return end
    local inicio = os.clock() -- ✅ CORRIGIDO: os.clock ao invés de tick()
    local conn
    conn = RunService.RenderStepped:Connect(function()
        if os.clock() - inicio >= duracao then -- ✅ CORRIGIDO
            conn:Disconnect()
            return
        end
        local p = 1 - ((os.clock() - inicio) / duracao) -- ✅ CORRIGIDO
        local sx = (math.random() - 0.5) * 2 * intensidade * p / 100
        local sy = (math.random() - 0.5) * 2 * intensidade * p / 100
        camera.CFrame = camera.CFrame * CFrame.new(sx, sy, 0)
    end)
end

local function fecharPortal(portalPart)
    if not portalPart or not portalPart.Parent then return end
    if portalPart:GetAttribute("Fechando") then return end
    portalPart:SetAttribute("Fechando", true)

    portalPart.CanTouch = false
    portalPart.CanCollide = false

    for _, emitter in ipairs(portalPart:GetChildren()) do
        if emitter:IsA("ParticleEmitter") then emitter.Enabled = false end
    end

    local somLoop = portalPart:FindFirstChild("PortalLoopSound")
    if somLoop and somLoop:IsA("Sound") then
        pcall(function()
            TweenService:Create(somLoop, TweenInfo.new(0.35, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Volume = 0}):Play()
        end)
    end

    local DUR = 0.45
    local cframeInicial = portalPart.CFrame

    TweenService:Create(portalPart, TweenInfo.new(DUR, Enum.EasingStyle.Quart, Enum.EasingDirection.In), {Size = Vector3.new(0, 0, 0.01)}):Play()

    local inicio = os.clock() -- ✅ CORRIGIDO
    local conn
    conn = RunService.Heartbeat:Connect(function()
        if not portalPart or not portalPart.Parent then
            if conn then conn:Disconnect() end
            return
        end
        local p = (os.clock() - inicio) / DUR -- ✅ CORRIGIDO
        if p >= 1 then
            conn:Disconnect()
            if portalPart and portalPart.Parent then portalPart:Destroy() end
            return
        end
        portalPart.CFrame = cframeInicial
    end)
end

local function wipePortalsPorTag(tag, comEfeito)
    local portais = {}
    local vistos = {}

    for _, p in ipairs(CollectionService:GetTagged(tag)) do
        if p and not vistos[p] then
            vistos[p] = true
            table.insert(portais, p)
        end
    end

    if comEfeito then
        for _, p in ipairs(portais) do
            if p and p.Parent then fecharPortal(p) end
        end
        if #portais > 0 then efeitoTP() end
    else
        for _, p in ipairs(portais) do
            if p and p.Parent then p:Destroy() end
        end
    end

    return #portais
end

local function wipeAllPortals(comEfeito)
    local total = wipePortalsPorTag(TAG_GEN, comEfeito)
    return total
end

    end
end

local function monitorarSaidaDoVoid()
    if voidMonitorConn then voidMonitorConn:Disconnect() end

    voidMonitorConn = RunService.Heartbeat:Connect(function()
        if not voidPlaceFolder then return end

        local char = player.Character
        if not char then
            voidForaDesde = nil
            return
        end

        local humanoid = char:FindFirstChildOfClass("Humanoid")
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end

        if humanoid and humanoid.Health <= 0 then
            voidForaDesde = nil
            task.defer(deletarVoidPlace)
            task.defer(deletarLocalTP)
            return
        end

        local pos = hrp.Position
        local estaFora = math.abs(pos.X) > VOID_TAMANHO + 20
            or math.abs(pos.Z) > VOID_TAMANHO + 20
            or pos.Y < VOID_Y - 100
            or pos.Y > VOID_Y + 500

        if estaFora then
            if not voidForaDesde then
                voidForaDesde = os.clock() -- ✅ CORRIGIDO
            end
            if os.clock() - voidForaDesde >= VOID_TOLERANCIA then -- ✅ CORRIGIDO
                voidForaDesde = nil
                task.defer(deletarVoidPlace)
                task.defer(deletarLocalTP)
            end
        else
            voidForaDesde = nil
        end
    end)
end

local function createPortalInstance(cframe, tag, opcoes)
    opcoes = opcoes or {}
    tag = tag or TAG_GEN
    local pos = cframe.Position
    local _, yRot, _ = cframe:ToOrientation()
    local verticalCFrame = CFrame.new(pos) * CFrame.Angles(0, yRot, 0)

    local portalPart = Instance.new("Part")
    portalPart.Size = Vector3.new(0.1, 0.1, 0.1)
    portalPart.CFrame = verticalCFrame
    portalPart.Anchored = true
    portalPart.CanCollide = false
    portalPart.Transparency = 1
    portalPart.Parent = workspace
    CollectionService:AddTag(portalPart, tag)
    -- ✅ REMOVIDO: table.insert(activePortals, portalPart) - memory leak

    if opcoes.comSom ~= false then
        local portalLoop = Instance.new("Sound")
        portalLoop.Name = "PortalLoopSound"
        portalLoop.SoundId = PORTAL_LOOP_SOUND_ID
        portalLoop.Volume = 2
        portalLoop.Looped = true
        portalLoop.RollOffMaxDistance = 150
        portalLoop.RollOffMode = Enum.RollOffMode.InverseTapered
        portalLoop.Parent = portalPart
        portalLoop:Play()
    end

    local portalLight = Instance.new("PointLight")
    portalLight.Color = Color3.fromRGB(50, 255, 80)
    portalLight.Range = 15
    portalLight.Brightness = 3
    portalLight.Parent = portalPart

    if opcoes.comParticulas ~= false then
        local sparks = Instance.new("ParticleEmitter")
        sparks.Color = ColorSequence.new(Color3.fromRGB(50, 255, 80))
        sparks.Size = NumberSequence.new({
            NumberSequenceKeypoint.new(0, 0.3),
            NumberSequenceKeypoint.new(1, 0)
        })
        sparks.Lifetime = NumberRange.new(0.5, 1.2)
        sparks.Rate = 25
        sparks.Speed = NumberRange.new(3, 7)
        sparks.SpreadAngle = Vector2.new(360, 360)
        sparks.EmissionDirection = Enum.NormalId.Front
        sparks.Parent = portalPart
    end

    local function createSurfaceFace(normalFace)
        local surf = Instance.new("SurfaceGui")
        surf.Face = normalFace
        surf.SizingMode = Enum.SurfaceGuiSizingMode.PixelsPerStud
        surf.PixelsPerStud = 50
        surf.LightInfluence = 0
        surf.Parent = portalPart

        local img = Instance.new("ImageLabel")
        img.Size = UDim2.new(1, 0, 1, 0)
        img.BackgroundTransparency = 1
        img.Image = PORTAL_TEXTURE
        img.AnchorPoint = Vector2.new(0.5, 0.5)
        img.Position = UDim2.new(0.5, 0, 0.5, 0)
        img.Parent = surf

        local c = Instance.new("UICorner")
        c.CornerRadius = UDim.new(0.5, 0)
        c.Parent = img
        return img
    end

    local fFront = createSurfaceFace(Enum.NormalId.Front)
    local bFront = createSurfaceFace(Enum.NormalId.Back)

    TweenService:Create(portalPart, TweenInfo.new(0.45, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Size = Vector3.new(7, 7, 0.1)}):Play()

    if opcoes.comGiro ~= false then
        task.spawn(function()
            while portalPart and portalPart.Parent do
                local fRot = (fFront.Rotation + 8) % 360
                fFront.Rotation = fRot
                bFront.Rotation = fRot
                task.wait(0.03)
            end
        end)
    end

    return portalPart, fFront, bFront
end

local function spawnPortalSaida(char)
    if not char or not char.Parent then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end

    local cframeAtual = root.CFrame
    local frente = cframeAtual.LookVector
    local costas = -frente

    local posicaoCostas = root.Position + costas * 2.5
    local cframeFinal = CFrame.new(posicaoCostas, posicaoCostas + frente)

    local portalAtras = createPortalInstance(cframeFinal, TAG_SAIDA, {
        comSom = false,
        comParticulas = false,
        comGiro = true,
    })

    if portalAtras then portalAtras.Transparency = 1 end

    task.delay(0.6, function()
        if portalAtras and portalAtras.Parent then fecharPortal(portalAtras) end
    end)

    return portalAtras
end

local function desconectarMarcacao()
    for _, conn in ipairs(marcacaoConns) do
        if conn and conn.Disconnect then pcall(function() conn:Disconnect() end) end
    end
    table.clear(marcacaoConns)
end

local function limparMarcacao(apagarPermanente)
    desconectarMarcacao()

    local manterP1Perm = permanenciaAtiva and (not apagarPermanente) and marcacaoP1Perm and marcacaoP1Perm.Parent

    if marcacaoP1 and marcacaoP1.Parent and marcacaoP1 ~= marcacaoP1Perm then
        fecharPortal(marcacaoP1)
    end
    if marcacaoP2 and marcacaoP2.Parent then
        fecharPortal(marcacaoP2)
    end

    if manterP1Perm then
        marcacaoP1 = marcacaoP1Perm
        marcacaoP2 = nil
    else
        if marcacaoP1Perm and marcacaoP1Perm.Parent then
            fecharPortal(marcacaoP1Perm)
        end
        marcacaoP1Perm = nil
        marcacaoP1 = nil
        marcacaoP2 = nil
    end

    atualizarLabelMarcacao()
end

    local isTeleporting = false
    local COOLDOWN = 1.0

    local function irPara(hit, destino)
        local char = player.Character
        if isTeleporting or not char or hit.Parent ~= char then return end
        if not destino or not destino.Parent then return end
        isTeleporting = true

        local root = char:FindFirstChild("HumanoidRootPart")
        if root then
            playTPSound()
            efeitoTP()
            task.spawn(function()
                task.wait(0.2)
                if root and root.Parent then
                    root.CFrame = CFrame.new(destino.Position + Vector3.new(0, 3, 0))
                    root.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                    root.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
                end
            end)
        end

        task.delay(0.5, function()
            desconectarMarcacao()

            if marcacaoP2 and marcacaoP2.Parent then
                fecharPortal(marcacaoP2)
            end
            marcacaoP2 = nil

            if permanenciaAtiva and marcacaoP1Perm and marcacaoP1Perm.Parent then
                marcacaoP1 = marcacaoP1Perm
            else
                if marcacaoP1 and marcacaoP1.Parent then
                    fecharPortal(marcacaoP1)
                end
                marcacaoP1 = nil
            end

            atualizarLabelMarcacao()

            task.delay(COOLDOWN, function()
                isTeleporting = false
            end)
        end)
    end

    table.insert(marcacaoConns, p1.Touched:Connect(function(h) irPara(h, p2) end))
    table.insert(marcacaoConns, p2.Touched:Connect(function(h) irPara(h, p1) end))
end

local function clicarMarcacao()
    local char = player.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then return end

    local state = marcacaoBtn:GetAttribute("ActiveState") or 0

    if state == 0 then
        setRandomQuote()
        desconectarMarcacao()

        if marcacaoP1 and marcacaoP1.Parent then fecharPortal(marcacaoP1) end
        if marcacaoP2 and marcacaoP2.Parent then fecharPortal(marcacaoP2) end
        marcacaoP1, marcacaoP2 = nil, nil

        if permanenciaAtiva then
            marcacaoP1Perm = createPortalInstance(root.CFrame * CFrame.new(0, 0, -5), TAG_M1)
            marcacaoP1 = marcacaoP1Perm
        else
            marcacaoP1 = createPortalInstance(root.CFrame * CFrame.new(0, 0, -5), TAG_M1)
        end

        atualizarLabelMarcacao()

    elseif state == 1 then
        setRandomQuote()
        if not marcacaoP1 or not marcacaoP1.Parent then

        atualizarLabelMarcacao()

        limparMarcacao(false)
    end

local function atualizarVisibilidade()
    local btns = {}
    for _, child in ipairs(scrollingList:GetChildren()) do
        if child:IsA("TextButton") then
            local nome = child:GetAttribute("NomeJogador") or ""
            local userId = child:GetAttribute("UserId") or 0
            local bate = (filtroAtivo == "") or (nome:lower():find(filtroAtivo:lower(), 1, true) ~= nil)

            child.Visible = bate
            if bate then
                table.insert(btns, {
                    btn = child,
                    isFav = favoritos[userId] == true,
                    nome = nome,
                })
            end
        end
    end

    table.sort(btns, function(a, b)
        if a.isFav ~= b.isFav then return a.isFav end
        return a.nome:lower() < b.nome:lower()
    end)

    for i, item in ipairs(btns) do
        item.btn.LayoutOrder = i
    end

    return #btns
end

local function refreshPlayerList()
    if refreshLock then return 0 end
    refreshLock = true

    local ok, resultado = pcall(function()
        -- 1. Limpa órfãos
        for _, child in ipairs(scrollingList:GetChildren()) do
            if child:IsA("TextButton") then
                local uid = child:GetAttribute("UserId")
                if not uid or not playerBtns[uid] or playerBtns[uid] ~= child then
                    child:Destroy()
                end
            end
        end

        -- 2. Limpa refs mortas
        for userId, btn in pairs(playerBtns) do
            if not btn or not btn.Parent then
                playerBtns[userId] = nil
            end
        end

        -- 3. Remove quem saiu
        for userId, btn in pairs(playerBtns) do
            local jogador = Players:GetPlayerByUserId(userId)
            if not jogador or jogador == player then
                if btn and btn.Parent then btn:Destroy() end
                playerBtns[userId] = nil
            end
        end

        -- 4. Pré-cache de amizades
        local amizades = {}
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= player then
                local okF, isF = pcall(function() return player:IsFriendsWith(p.UserId) end)
                amizades[p.UserId] = okF and isF or false
            end
        end

        -- 5. Cria os que faltam
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= player and not playerBtns[p.UserId] then
                playerBtns[p.UserId] = criarBotaoPlayer(p, 0, amizades[p.UserId] or false)
            end
        end

        local count = atualizarVisibilidade()
        scrollingList.CanvasSize = UDim2.new(0, 0, 0, count * 19)
        return count
    end)

    refreshLock = false

    if not ok then
        warn("[RickPrime] Erro no refreshPlayerList:", resultado)
        return 0
    end

    return resultado
end

    end
    return count
end

local randomTPAtivo = false
local randomPortal = nil

local function encontrarPosicaoAleatoriaValida()
    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    
    local posicaoAtual = hrp.Position
    local distanciaMinima = 500
    local distanciaMaxima = 2000
    local melhorPosicao = nil
    local melhorDistancia = 0
    
    local tentativas = 0
    local maxTentativas = 100
    
    while tentativas < maxTentativas do
        tentativas += 1
        
        -- ✅ Previne lag spike
        if tentativas % 20 == 0 then task.wait() end
        
        -- Gera posição aleatória (somente X e Z)
        local angulo = math.random() * math.pi * 2
        local distancia = math.random(distanciaMinima, distanciaMaxima)
        local x = posicaoAtual.X + math.cos(angulo) * distancia
        local z = posicaoAtual.Z + math.sin(angulo) * distancia
        
        local posicaoTeste = Vector3.new(x, posicaoAtual.Y + 500, z) -- Começa do alto
        
        -- Raycast para encontrar o chão
        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        params.FilterDescendantsInstances = { char }
        params.IgnoreWater = false
        
        local resultado = workspace:Raycast(posicaoTeste, Vector3.new(0, -1000, 0), params)
        
        if resultado then
            local posicaoChao = resultado.Position + Vector3.new(0, 5, 0)
            local distanciaReal = (posicaoAtual - Vector3.new(posicaoChao.X, posicaoAtual.Y, posicaoChao.Z)).Magnitude
            
            -- Verifica se não é muito perto e tem chão válido
            if distanciaReal >= distanciaMinima then
                return posicaoChao
            elseif distanciaReal > melhorDistancia then
                melhorDistancia = distanciaReal
                melhorPosicao = posicaoChao
            end
        end
    end
    
    -- Se não achou a 500+, retorna o mais longe que encontrou
    return melhorPosicao
end

local function avBuscarSuperficie(posicao, character)
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = { character }
    params.IgnoreWater = false

    local candidatos = {}
    local raio = AV.RaioInicial
    local raycastCount = 0 -- ✅ CONTADOR DE SEGURANÇA

    while raio <= AV.RaioMaximo do
        for i = 0, AV.PassosPorAnel - 1 do
            raycastCount += 1
            -- ✅ PREVINE LAG SPIKE
            if raycastCount % 200 == 0 then task.wait() end
            
            local angulo = (i / AV.PassosPorAnel) * math.pi * 2
            local x = posicao.X + math.cos(angulo) * raio
            local z = posicao.Z + math.sin(angulo) * raio
            local origem = Vector3.new(x, posicao.Y + AV.AlturaAcima, z)
            local direcao = Vector3.new(0, -(AV.AlturaAcima + AV.AlturaAbaixo), 0)
            local res = workspace:Raycast(origem, direcao, params)
            
            if res then
                local dot = res.Normal:Dot(Vector3.new(0, 1, 0))
                if dot >= AV.DotMinimoPisavel then
                    table.insert(candidatos, { posicao = res.Position, raio = raio })
                end
            end
        end
        if #candidatos > 0 and raio >= AV.RaioInicial * 4 then break end
        raio = raio + AV.RaioIncremento
    end

    if #candidatos == 0 then return nil end

    local melhor = candidatos[1]
    for _, c in ipairs(candidatos) do
        local dY = c.posicao.Y - melhor.posicao.Y
        if dY > AV.ToleranciaY then
            melhor = c
        elseif math.abs(dY) <= AV.ToleranciaY then
            if c.raio < melhor.raio then melhor = c end
        end
    end
    return melhor.posicao
end

local function avTeleportar(posicao, character)
    local hrp = character:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    local rot = hrp.CFrame - hrp.CFrame.Position
    hrp.CFrame = CFrame.new(posicao) * rot
    hrp.AssemblyLinearVelocity  = Vector3.new(0, 0, 0)
    hrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
end

local function avTeleportarParaSuperficie(character)
    if avTpCooldown then return end
    avTpCooldown = true
    getgenv().AntiTP_IgnorarProximo = true

    avTempoCaindo = 0
    avUltimoTempo = os.clock() -- ✅ CORRIGIDO

    local sucesso, erro = pcall(function()
        local hrp = character:FindFirstChild("HumanoidRootPart")
        if not hrp then return end

        avEfeitoTeleporte()

        local superficie = avBuscarSuperficie(hrp.Position, character)
        if superficie then
            avTeleportar(superficie + Vector3.new(0, AV.OffsetY, 0), character)
        elseif avUltimaPosicaoSegura then
            avTeleportar(avUltimaPosicaoSegura + Vector3.new(0, AV.OffsetY, 0), character)
        else
            local spawn = workspace:FindFirstChildOfClass("SpawnLocation")
            if spawn then
                avTeleportar(spawn.Position + Vector3.new(0, 5, 0), character)
            end
        end

        avTempoCaindo = 0
        avUltimoTempo = os.clock() -- ✅ CORRIGIDO
        avUltimaPos = nil
    end)

    if not sucesso then warn("[AntiVoid] erro no TP:", erro) end

    local cooldown = avLocalSeguroAtivo and 0.3 or 1.5
    task.wait(cooldown)
    avTpCooldown = false
end

local function avRegistrarMorte()
    local agora = os.clock() -- ✅ CORRIGIDO
    table.insert(avHistoricoMortes, agora)

    for i = #avHistoricoMortes, 1, -1 do
        if agora - avHistoricoMortes[i] > AV.JanelaMortes then
            table.remove(avHistoricoMortes, i)
        end
    end

    if #avHistoricoMortes >= AV.MortesLimite then
        avHistoricoMortes = {}
        task.spawn(function()
            task.wait(0.5)
            local char = player.Character
            if not char then return end
            local hrp = char:FindFirstChild("HumanoidRootPart")
            if not hrp then return end

            avEfeitoTeleporte()
            local superficie = avBuscarSuperficie(hrp.Position, char)
            if superficie then
                avTeleportar(superficie + Vector3.new(0, AV.OffsetY, 0), char)
            elseif avUltimaPosicaoSegura then
                avTeleportar(avUltimaPosicaoSegura + Vector3.new(0, AV.OffsetY, 0), char)
            else
                local spawn = workspace:FindFirstChildOfClass("SpawnLocation")
                if spawn then
                    avTeleportar(spawn.Position + Vector3.new(0, 5, 0), char)
                end
            end
            avLocalSeguroAtivo = true
            avLocalSeguroExpira = os.clock() + AV.LocalSeguroDuracao -- ✅ CORRIGIDO
        end)
    end
end

RunService.Heartbeat:Connect(function()
    if not antiVoidAtivo then return end

    local character = player.Character
    if not character then
        avTempoCaindo = 0
        avUltimaPos = nil
        return
    end

    local hrp      = character:FindFirstChild("HumanoidRootPart")
    local humanoid = character:FindFirstChildOfClass("Humanoid")

    if not hrp or not humanoid then
        avTempoCaindo = 0
        avUltimaPos = nil
        return
    end
    if humanoid.Health <= 0 then
        avTempoCaindo = 0
        avUltimaPos = nil
        return
    end

    if avDesativado and os.clock() < avDesativadoExpira then -- ✅ CORRIGIDO
        avTempoCaindo = 0
        avUltimaPos = nil
        return
    end

    if hrp.Position.Y > 50 then
        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        params.FilterDescendantsInstances = { character }
        local origem = hrp.Position + Vector3.new(0, 2, 0)
        local direcao = Vector3.new(0, -15, 0)
        if workspace:Raycast(origem, direcao, params) then
            avUltimaPosicaoSegura = hrp.Position
        end
    end

    local agora = os.clock() -- ✅ CORRIGIDO
    local dt = agora - avUltimoTempo
    avUltimoTempo = agora

    local velocidadeY = hrp.AssemblyLinearVelocity.Y
    local caindoPorVel = velocidadeY < AV.VelocidadeMinimaCaindo
    local state = humanoid:GetState()
    local caindoPorEstado = (state == Enum.HumanoidStateType.Freefall)
        or (state == Enum.HumanoidStateType.FallingDown)

    local caindoPorPos = false
    if avUltimaPos then
        local deltaY = hrp.Position.Y - avUltimaPos.Y
        caindoPorPos = (deltaY < -0.5) and (velocidadeY < -5)
    end
    avUltimaPos = hrp.Position

    if velocidadeY > 5 then
        avTempoCaindo = 0
        return
    end

    local estaCaindo = caindoPorVel or caindoPorEstado or caindoPorPos
    local limiteAtual = (avLocalSeguroAtivo and os.clock() < avLocalSeguroExpira) -- ✅ CORRIGIDO
        and AV.TempoCaindoLimite or AV.TempoCaindoNormal

    if estaCaindo then
        avTempoCaindo = avTempoCaindo + dt
        if avTempoCaindo >= limiteAtual and not avTpCooldown then
            avTeleportarParaSuperficie(character)
        end
    else
        avTempoCaindo = 0
    end
end)

    tpPlayerBtn, clearBtn, voidBtn, ceusBtn, randomBtn, -- ✅ ADICIONADO randomBtn
    marcacaoBtn, permBtn, antiTPBtn, antiVoidBtn, rejoinBtn, closeBtn
}

minimizarBtn.MouseButton1Click:Connect(function()
    setRandomQuote()
    fecharPainel()
    menuMinimizado = not menuMinimizado

    local lbl = minimizarBtn:FindFirstChildOfClass("TextLabel")

    if menuMinimizado then
        if lbl then
            lbl.Text = "▸ EXPANDIR"
            lbl.TextColor3 = COR_CIANO
        end
        minimizarBtn.BackgroundColor3 = Color3.fromRGB(10, 35, 40)
        if minimizarStroke then
            minimizarStroke.Color = COR_CIANO
            minimizarStroke.Transparency = 0.1
        end

        TweenService:Create(titleTag, TweenInfo.new(0.25), {TextTransparency = 1}):Play()
        TweenService:Create(title, TweenInfo.new(0.25), {TextTransparency = 1}):Play()
        TweenService:Create(liveDot, TweenInfo.new(0.25), {BackgroundTransparency = 1}):Play()
        TweenService:Create(liveDotStroke, TweenInfo.new(0.25), {Transparency = 1}):Play()
        TweenService:Create(quoteLabel, TweenInfo.new(0.25), {TextTransparency = 1}):Play()
        TweenService:Create(divTop, TweenInfo.new(0.25), {BackgroundTransparency = 1}):Play()
        TweenService:Create(footerLabel, TweenInfo.new(0.25), {TextTransparency = 1}):Play()
        TweenService:Create(statusBar, TweenInfo.new(0.25), {BackgroundTransparency = 1}):Play()
        TweenService:Create(innerStroke, TweenInfo.new(0.25), {Transparency = 1}):Play()

        for _, btn in ipairs(ocultarAoMinimizar) do
            if btn and btn.Parent then
                btn.Visible = false
                btn.Active = false
                btn.Selectable = false
            end
        end

        minimizarBtn.Position = UDim2.new(0.05, 0, 0, 10)

        TweenService:Create(mainFrame, TweenInfo.new(0.35, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {
            Size = UDim2.new(0, 100, 0, alturaMinimizada)
        }):Play()

    else
        if lbl then
            lbl.Text = "▾ MINIMIZAR"
            lbl.TextColor3 = COR_TEXTO
        end
        minimizarBtn.BackgroundColor3 = COR_PANEL
        if minimizarStroke then
            minimizarStroke.Color = COR_CIANO
            minimizarStroke.Transparency = 0.4
        end

        TweenService:Create(titleTag, TweenInfo.new(0.25), {TextTransparency = 0}):Play()
        TweenService:Create(title, TweenInfo.new(0.25), {TextTransparency = 0}):Play()
        TweenService:Create(liveDot, TweenInfo.new(0.25), {BackgroundTransparency = 0}):Play()
        TweenService:Create(liveDotStroke, TweenInfo.new(0.25), {Transparency = 0.5}):Play()
        TweenService:Create(quoteLabel, TweenInfo.new(0.25), {TextTransparency = 0}):Play()
        TweenService:Create(divTop, TweenInfo.new(0.25), {BackgroundTransparency = 0}):Play()
        TweenService:Create(footerLabel, TweenInfo.new(0.25), {TextTransparency = 0}):Play()
        TweenService:Create(statusBar, TweenInfo.new(0.25), {BackgroundTransparency = 0}):Play()
        TweenService:Create(innerStroke, TweenInfo.new(0.25), {Transparency = 0.4}):Play()

        minimizarBtn.Position = UDim2.new(0.05, 0, 0, 224) -- ✅ ATUALIZADO

        TweenService:Create(mainFrame, TweenInfo.new(0.35, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {
            Size = UDim2.new(0, 100, 0, alturaOriginal)
        }):Play()

        for _, btn in ipairs(ocultarAoMinimizar) do
            if not btn or not btn.Parent then continue end
            btn.Visible = true
            btn.Active = true
            btn.Selectable = true

            if btn == ceusBtn then
                btn.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
                btn.BackgroundTransparency = 0

                local l = btn:FindFirstChildOfClass("TextLabel")
                if l then
                    l.TextTransparency = 0
                    l.TextColor3 = COR_CIANO
                end

                local st = btn:FindFirstChildOfClass("UIStroke")
                if st then
                    st.Transparency = 0.2
                    st.Color = COR_CIANO
                end

                local a = btn:FindFirstChild("Frame")
                if a and a:IsA("Frame") then
                    a.BackgroundTransparency = 0
                    a.BackgroundColor3 = COR_CIANO
                end
            elseif btn == randomBtn then -- ✅ NOVO BOTÃO
                btn.BackgroundColor3 = COR_PANEL
                btn.BackgroundTransparency = 0.15

                local l = btn:FindFirstChildOfClass("TextLabel")
                if l then
                    l.TextTransparency = 0
                    l.TextColor3 = Color3.fromRGB(255, 100, 180)
                end

                local st = btn:FindFirstChildOfClass("UIStroke")
                if st then
                    st.Transparency = 0.3
                    st.Color = Color3.fromRGB(255, 100, 180)
                end

                local a = btn:FindFirstChild("Frame")
                if a and a:IsA("Frame") then
                    a.BackgroundTransparency = 0
                    a.BackgroundColor3 = Color3.fromRGB(255, 100, 180)
                end
            else
                local transOriginal = 0.15
                local strokeTransOriginal = 0.5

                if btn == closeBtn then
                    strokeTransOriginal = 0.2
                elseif btn == voidBtn then
                    strokeTransOriginal = 0.3
                elseif btn == antiTPBtn or btn == marcacaoBtn or btn == permBtn or btn == antiVoidBtn or btn == rejoinBtn then
                    strokeTransOriginal = 0.4
                end

                btn.BackgroundTransparency = transOriginal
                btn.BackgroundColor3 = COR_PANEL

                local l = btn:FindFirstChildOfClass("TextLabel")
                if l then
                    l.TextTransparency = 0
                    l.TextColor3 = COR_TEXTO
                end

                local st = btn:FindFirstChildOfClass("UIStroke")
                if st then st.Transparency = strokeTransOriginal end

                local a = btn:FindFirstChild("Frame")
                if a and a:IsA("Frame") then a.BackgroundTransparency = 0 end
            end
        end
    end
end)

-- CLEANUP
--=============================================================
local function fullCleanup()
    getgenv().AntiTP_Ativo = false
    getgenv().Marcacao_Ativo = false
    antiVoidAtivo = false
    randomTPAtivo = false -- ✅ NOVO

    wipePortalsPorTag(TAG_GEN, true)
    wipePortalsPorTag(TAG_M1, true)
    wipePortalsPorTag(TAG_M2, true)
    wipePortalsPorTag(TAG_VOID, true)
    wipePortalsPorTag(TAG_RET, true)
    wipePortalsPorTag(TAG_SAIDA, true)

    deletarVoidPlace()
    deletarLocalTP()
    deletarCeuPlace()
    deletarCeuLocalTP()

    limparMarcacao(true)

    permanenciaAtiva = false
    marcacaoP1Perm = nil

    if blurEffect and blurEffect.Parent then
        blurEffect:Destroy()
        blurEffect = nil
    end
    blurTween = nil

    for _, s in ipairs(SoundService:GetChildren()) do
        if s:IsA("Sound") then pcall(function() s:Destroy() end) end
    end

    if screenGui and screenGui.Parent then screenGui:Destroy() end
    if fxGui and fxGui.Parent then fxGui:Destroy() end

    getgenv().RickPortalCleanup = nil
end

getgenv().RickPortalCleanup = fullCleanup

closeBtn.MouseButton1Click:Connect(function()
    setRandomQuote()

    if closeLbl then
        closeLbl.Text = "✕ FECHANDO..."
        closeLbl.TextColor3 = COR_VERMELHO
    end

    getgenv().AntiTP_Ativo = false
    getgenv().Marcacao_Ativo = false
    antiVoidAtivo = false
    randomTPAtivo = false -- ✅ NOVO

    wipeAllPortals(true)
    limparMarcacao(true)

    deletarVoidPlace()
    deletarLocalTP()
    deletarCeuPlace()
    deletarCeuLocalTP()

    if panelTween then
        panelTween:Cancel()
        panelTween = nil
    end
    panelOpen = false

    local slideOutTween = TweenService:Create(
        mainFrame,
        TweenInfo.new(0.8, Enum.EasingStyle.Sine, Enum.EasingDirection.In),
        {Position = UDim2.new(0, -150, 0.35, -178)} -- ✅ ATUALIZADO
    )
    slideOutTween:Play()
    slideOutTween.Completed:Wait()

    fullCleanup()
end)
