local function criarGradiente(parent, c1, c2, rotacao, transparencia)
    local grad = Instance.new("UIGradient")
    grad.Color = ColorSequence.new(c1, c2)
    grad.Rotation = rotacao or 0

    end
    grad.Parent = parent
    return grad

-- ✨ Brilho pulsante no stroke
local function criarBrilhoPulsante(parent, cor, minMs, maxMs)
    task.spawn(function()
        local stroke = parent:FindFirstChildOfClass("UIStroke")
        if not stroke then return end
        local corOriginal = stroke.Color
        local transOriginal = stroke.Transparency
        while parent and parent.Parent do
            TweenService:Create(stroke, TweenInfo.new(math.random(minMs, maxMs) / 100, Enum.EasingStyle.Sine), {
                Transparency = math.max(0, transOriginal - 0.3),
                Color = cor
            }):Play()
            task.wait(math.random(minMs, maxMs) / 100)
            if not (parent and parent.Parent) then break end
            TweenService:Create(stroke, TweenInfo.new(math.random(minMs, maxMs) / 100, Enum.EasingStyle.Sine), {
                Transparency = transOriginal,
                Color = corOriginal
            }):Play()
            task.wait(math.random(minMs, maxMs) / 100)
        end
    end)
end

local blurEffect = nil
local blurTween = nil

local function criarBlur(tamanho)
    if blurEffect and blurEffect.Parent then blurEffect:Destroy() end
    if blurTween then blurTween:Cancel() blurTween = nil end
    blurEffect = Instance.new("BlurEffect")
    blurEffect.Name = "RP_Blur"
    blurEffect.Size = 0
    blurEffect.Parent = Lighting
    blurTween = TweenService:Create(blurEffect, TweenInfo.new(0.1, Enum.EasingStyle.Quad), {Size = tamanho})
    blurTween:Play()
end

local function removerBlur(duracao)
    if not blurEffect then return end
    local b = blurEffect
    if blurTween then blurTween:Cancel() blurTween = nil end
    blurTween = TweenService:Create(b, TweenInfo.new(duracao or 0.4, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Size = 0})
    blurTween:Play()
    task.delay(duracao or 0.4, function()
        if b and b.Parent then b:Destroy() end
        if blurEffect == b then blurEffect = nil blurTween = nil end
    end)
end

local function cameraShake(intensidade, duracao)
    local camera = workspace.CurrentCamera
    if not camera then return end
    local inicio = os.clock()
    local conn
    conn = RunService.RenderStepped:Connect(function()
        local decorrido = os.clock() - inicio
        if decorrido >= duracao then
            conn:Disconnect()
            return
        end
        local p = 1 - (decorrido / duracao)
        local sx = (math.random() - 0.5) * 2 * intensidade * p / 100
        local sy = (math.random() - 0.5) * 2 * intensidade * p / 100
        camera.CFrame = camera.CFrame * CFrame.new(sx, sy, 0)
    end)
end

local function fecharPortal(portalPart)
    if not portalPart or not portalPart.Parent then return end
    if portalPart:GetAttribute("Fechando") then return end
    portalPart:SetAttribute("Fechando", true)

    portalPart.CanTouch = false
    portalPart.CanCollide = false

    for _, emitter in ipairs(portalPart:GetChildren()) do
        if emitter:IsA("ParticleEmitter") then emitter.Enabled = false end
        if emitter:IsA("Light") then
            pcall(function()
                TweenService:Create(emitter, TweenInfo.new(0.35), {Brightness = 0}):Play()
            end)
        end
    end

    -- Limpa elementos extras
    for _, filho in ipairs(portalPart:GetChildren()) do
        if filho.Name == "RingRotativo" or filho.Name == "LightBeam" then
            pcall(function()
                if filho:IsA("BasePart") then
                    TweenService:Create(filho, TweenInfo.new(0.4), {Transparency = 1}):Play()
                    task.delay(0.4, function() if filho and filho.Parent then filho:Destroy() end end)
                end
            end)
        end
    end

    local somLoop = portalPart:FindFirstChild("PortalLoopSound")
    if somLoop and somLoop:IsA("Sound") then
        pcall(function()
            TweenService:Create(somLoop, TweenInfo.new(0.35, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Volume = 0}):Play()
        end)
    end

    local DUR = 0.5
    local cframeInicial = portalPart.CFrame

    TweenService:Create(portalPart, TweenInfo.new(DUR, Enum.EasingStyle.Back, Enum.EasingDirection.In), {
        Size = Vector3.new(0, 0, 0.01)
    }):Play()

    -- ✨ Explosão de partículas no fechamento
    task.spawn(function()
        local pAux = Instance.new("Part")
        pAux.Anchored = true
        pAux.CanCollide = false
        pAux.Transparency = 1
        pAux.Size = Vector3.new(1, 1, 1)
        pAux.CFrame = cframeInicial
        pAux.Parent = workspace

        local pe = Instance.new("ParticleEmitter")
        pe.Color = ColorSequence.new(COR_LINHA, Color3.fromRGB(255, 255, 255))
        pe.Lifetime = NumberRange.new(0.3, 0.7)
        pe.Rate = 0
        pe.Speed = NumberRange.new(10, 20)

local function wipePortalsPorTag(tag, comEfeito)
    local portais = {}
    local vistos = {}
    for _, p in ipairs(CollectionService:GetTagged(tag)) do
        if p and not vistos[p] then
            vistos[p] = true
            table.insert(portais, p)
        end
    end
    if comEfeito then
        for _, p in ipairs(portais) do
            if p and p.Parent then fecharPortal(p) end
        end
        if #portais > 0 then efeitoTP() end
    else
        for _, p in ipairs(portais) do
            if p and p.Parent then p:Destroy() end
        end
    end
    return #portais
end

local function wipeAllPortals(comEfeito)
    local total = wipePortalsPorTag(TAG_GEN, comEfeito)
    table.clear(activePortals)
    return total
end

local function monitorarSaidaDoVoid()
    if voidMonitorConn then voidMonitorConn:Disconnect() end
    voidMonitorConn = RunService.Heartbeat:Connect(function()
        if not voidPlaceFolder then return end
        local char = player.Character
        if not char then voidForaDesde = nil return end
        local humanoid = char:FindFirstChildOfClass("Humanoid")
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end
        if humanoid and humanoid.Health <= 0 then
            voidForaDesde = nil
            task.defer(deletarVoidPlace)
            task.defer(deletarLocalTP)
            return
        end
        local pos = hrp.Position
        local estaFora = math.abs(pos.X) > VOID_TAMANHO + 20
            or math.abs(pos.Z) > VOID_TAMANHO + 20
            or pos.Y < VOID_Y - 100
            or pos.Y > VOID_Y + 500
        if estaFora then
            if not voidForaDesde then voidForaDesde = os.clock() end
            if os.clock() - voidForaDesde >= VOID_TOLERANCIA then
                voidForaDesde = nil
                task.defer(deletarVoidPlace)
                task.defer(deletarLocalTP)
            end
        else voidForaDesde = nil end
    end)
end

local function createPortalInstance(cframe, tag, opcoes)
    opcoes = opcoes or {}
    tag = tag or TAG_GEN
    local pos = cframe.Position
    local _, yRot, _ = cframe:ToOrientation()
    local verticalCFrame = CFrame.new(pos) * CFrame.Angles(0, yRot, 0)

    local portalPart = Instance.new("Part")
    portalPart.Size = Vector3.new(0.1, 0.1, 0.1)
    portalPart.CFrame = verticalCFrame
    portalPart.Anchored = true
    portalPart.CanCollide = false
    portalPart.Transparency = 1
    portalPart.Parent = workspace
    CollectionService:AddTag(portalPart, tag)
    if tag == TAG_GEN then table.insert(activePortals, portalPart) end

    if opcoes.comSom ~= false then
        local portalLoop = Instance.new("Sound")
        portalLoop.Name = "PortalLoopSound"
        portalLoop.SoundId = PORTAL_LOOP_SOUND_ID
        portalLoop.Volume = 2
        portalLoop.Looped = true
        portalLoop.RollOffMaxDistance = 150
        portalLoop.RollOffMode = Enum.RollOffMode.InverseTapered
        portalLoop.Parent = portalPart
        portalLoop:Play()
    end

    -- ✨ PointLight + SpotLight
    local portalLight = Instance.new("PointLight")
    portalLight.Color = COR_LINHA
    portalLight.Range = 25
    portalLight.Brightness = 5
    portalLight.Parent = portalPart

    local spotlight = Instance.new("SpotLight")
    spotlight.Color = COR_LINHA
    spotlight.Range = 30
    spotlight.Brightness = 3
    spotlight.Angle = 90
    spotlight.Face = Enum.NormalId.Front
    spotlight.Parent = portalPart

    -- ✨ Múltiplas camadas de partículas
    if opcoes.comParticulas ~= false then
        local sparks = Instance.new("ParticleEmitter")
        sparks.Name = "PortalSparks"
        sparks.Color = ColorSequence.new(COR_LINHA)
        sparks.Size = NumberSequence.new({
            NumberSequenceKeypoint.new(0, 0.5),
            NumberSequenceKeypoint.new(1, 0)
        })
        sparks.Lifetime = NumberRange.new(0.6, 1.4)
        sparks.Rate = 40
        sparks.Speed = NumberRange.new(5, 12)
        sparks.SpreadAngle = Vector2.new(360, 360)
        sparks.EmissionDirection = Enum.NormalId.Front
        sparks.LightEmission = 1
        sparks.Parent = portalPart

        local brasas = Instance.new("ParticleEmitter")
        brasas.Name = "PortalBrasas"
        brasas.Color = ColorSequence.new(COR_OURO)
        brasas.Size = NumberSequence.new({
            NumberSequenceKeypoint.new(0, 0.3),
            NumberSequenceKeypoint.new(0.5, 0.5),
            NumberSequenceKeypoint.new(1, 0)
        })
        brasas.Lifetime = NumberRange.new(1, 2.5)
        brasas.Rate = 25
        brasas.Speed = NumberRange.new(3, 8)
        brasas.SpreadAngle = Vector2.new(30, 30)
        brasas.Acceleration = Vector3.new(0, 10, 0)
        brasas.LightEmission = 0.8
        brasas.Parent = portalPart

        local glowP = Instance.new("ParticleEmitter")
        glowP.Name = "PortalGlow"
        glowP.Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, Color3.fromRGB(200, 255, 200)),
            ColorSequenceKeypoint.new(1, COR_LINHA)
        })
        glowP.Size = NumberSequence.new({
            NumberSequenceKeypoint.new(0, 1),
            NumberSequenceKeypoint.new(1, 2)
        })
        glowP.Transparency = NumberSequence.new({
            NumberSequenceKeypoint.new(0, 0.5),
            NumberSequenceKeypoint.new(1, 1)
        })
        glowP.Lifetime = NumberRange.new(0.8, 1.2)
        glowP.Rate = 15
        glowP.Speed = NumberRange.new(0.5, 2)
        glowP.SpreadAngle = Vector2.new(360, 360)
        glowP.LightEmission = 0.5
        glowP.Parent = portalPart
    end

    -- ✨ Anel rotativo
    local ring = Instance.new("Part")
    ring.Name = "RingRotativo"
    ring.Shape = Enum.PartType.Cylinder
    ring.Size = Vector3.new(0.15, 5, 5)
    ring.CFrame = verticalCFrame * CFrame.Angles(0, 0, math.rad(90))
    ring.Anchored = true
    ring.CanCollide = false
    ring.Material = Enum.Material.Neon
    ring.Color = COR_LINHA
    ring.Transparency = 0.3
    ring.Parent = workspace
    CollectionService:AddTag(ring, tag .. "_Ring")

    task.spawn(function()
        while ring and ring.Parent do
            ring.CFrame = ring.CFrame * CFrame.Angles(math.rad(2), 0, 0)
            task.wait(0.03)
        end
    end)

    -- ✨ Feixe de luz vertical
    local beam = Instance.new("Part")
    beam.Name = "LightBeam"
    beam.Size = Vector3.new(1, 50, 1)
    beam.CFrame = CFrame.new(pos + Vector3.new(0, 25, 0))
    beam.Anchored = true
    beam.CanCollide = false
    beam.Material = Enum.Material.Neon
    beam.Color = COR_LINHA
    beam.Transparency = 0.85
    beam.Parent = workspace
    CollectionService:AddTag(beam, tag .. "_Beam")

    task.spawn(function()
        while beam and beam.Parent do
            beam.Transparency = 0.7 + math.sin(tick() * 3) * 0.15
            task.wait(0.05)
        end
    end)

        local surf = Instance.new("SurfaceGui")
        surf.Face = normalFace

        stroke.Color = COR_LINHA
        stroke.Thickness = 3
        stroke.Transparency = 0.2

        Size = Vector3.new(7, 7, 0.1)
    }):Play()
    TweenService:Create(ring, TweenInfo.new(0.5, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
        Size = Vector3.new(0.15, 8, 8)
    }):Play()

    -- ✨ Pulso na luz
    task.spawn(function()
        while portalPart and portalPart.Parent do
            TweenService:Create(portalLight, TweenInfo.new(0.6, Enum.EasingStyle.Sine), {Brightness = 7}):Play()
            task.wait(0.6)
            if not (portalPart and portalPart.Parent) then break end
            TweenService:Create(portalLight, TweenInfo.new(0.6, Enum.EasingStyle.Sine), {Brightness = 4}):Play()
            task.wait(0.6)
        end
    end)

    if opcoes.comGiro ~= false then
        task.spawn(function()
            while portalPart and portalPart.Parent do
                local fRot = (fFront.Rotation + 10) % 360
                fFront.Rotation = fRot
                bFront.Rotation = fRot
                task.wait(0.03)
            end
        end)
    end

    return portalPart, fFront, bFront
end

local function spawnPortalSaida(char)
    if not char or not char.Parent then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end

    local cframeAtual = root.CFrame
    local frente = cframeAtual.LookVector
    local costas = -frente

    local posicaoCostas = root.Position + costas * 2.5
    local cframeFinal = CFrame.new(posicaoCostas, posicaoCostas + frente)

    local portalAtras = createPortalInstance(cframeFinal, TAG_SAIDA, {
        comSom = false,
        comParticulas = false,
        comGiro = true,
    })

    if portalAtras then portalAtras.Transparency = 1 end

    task.delay(0.8, function()
        if portalAtras and portalAtras.Parent then fecharPortal(portalAtras) end
    end)

    return portalAtras
end

local function desconectarMarcacao()
    for _, conn in ipairs(marcacaoConns) do
        if conn and conn.Disconnect then pcall(function() conn:Disconnect() end) end
    end
    table.clear(marcacaoConns)
end

local function limparMarcacao(apagarPermanente)
    desconectarMarcacao()
    local manterP1Perm = permanenciaAtiva and (not apagarPermanente) and marcacaoP1Perm and marcacaoP1Perm.Parent

    if marcacaoP1 and marcacaoP1.Parent and marcacaoP1 ~= marcacaoP1Perm then
        fecharPortal(marcacaoP1)
    end
    if marcacaoP2 and marcacaoP2.Parent then
        fecharPortal(marcacaoP2)
    end

    if manterP1Perm then
        marcacaoP1 = marcacaoP1Perm
        marcacaoP2 = nil
    else
        if marcacaoP1Perm and marcacaoP1Perm.Parent then
            fecharPortal(marcacaoP1Perm)
        end
        marcacaoP1Perm = nil
        marcacaoP1 = nil
        marcacaoP2 = nil
    end

    atualizarLabelMarcacao()
end

local function conectarPortais(p1, p2)
    local isTeleporting = false
    local COOLDOWN = 1.0

    local function irPara(hit, destino)
        local char = player.Character
        if isTeleporting or not char or hit.Parent ~= char then return end
        if not destino or not destino.Parent then return end
        isTeleporting = true

        local root = char:FindFirstChild("HumanoidRootPart")
        if root then
            playTPSound()
            efeitoTP()
            task.spawn(function()
                task.wait(0.2)
                if root and root.Parent then
                    root.CFrame = CFrame.new(destino.Position + Vector3.new(0, 3, 0))
                    root.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                    root.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
                end
            end)
        end

        task.delay(0.5, function()
            desconectarMarcacao()
            if marcacaoP2 and marcacaoP2.Parent then
                fecharPortal(marcacaoP2)
            end
            marcacaoP2 = nil

            if permanenciaAtiva and marcacaoP1Perm and marcacaoP1Perm.Parent then
                marcacaoP1 = marcacaoP1Perm
            else
                if marcacaoP1 and marcacaoP1.Parent then
                    fecharPortal(marcacaoP1)
                end
                marcacaoP1 = nil
            end

            atualizarLabelMarcacao()
            task.delay(COOLDOWN, function()
                isTeleporting = false
            end)
        end)
    end

    table.insert(marcacaoConns, p1.Touched:Connect(function(h) irPara(h, p2) end))
    table.insert(marcacaoConns, p2.Touched:Connect(function(h) irPara(h, p1) end))
end

local function clicarMarcacao()
    local char = player.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then return end

    local state = marcacaoBtn:GetAttribute("ActiveState") or 0

    if state == 0 then
        setRandomQuote()
        desconectarMarcacao()
        if marcacaoP1 and marcacaoP1.Parent then fecharPortal(marcacaoP1) end
        if marcacaoP2 and marcacaoP2.Parent then fecharPortal(marcacaoP2) end
        marcacaoP1, marcacaoP2 = nil, nil

        if permanenciaAtiva then
            marcacaoP1Perm = createPortalInstance(root.CFrame * CFrame.new(0, 0, -5), TAG_M1)
            marcacaoP1 = marcacaoP1Perm
        else
            marcacaoP1 = createPortalInstance(root.CFrame * CFrame.new(0, 0, -5), TAG_M1)
        end
        atualizarLabelMarcacao()

    elseif state == 1 then
        setRandomQuote()
        if not marcacaoP1 or not marcacaoP1.Parent then
            limparMarcacao()
            return
        end
        marcacaoP2 = createPortalInstance(root.CFrame * CFrame.new(0, 0, -5), TAG_M2)
        atualizarLabelMarcacao()
        conectarPortais(marcacaoP1, marcacaoP2)

    elseif state == 2 then
        setRandomQuote()
        limparMarcacao(false)
    end
end

local function atualizarVisibilidade()
    local btns = {}
    for _, child in ipairs(scrollingList:GetChildren()) do
        if child:IsA("TextButton") then
            local nome = child:GetAttribute("NomeJogador") or ""
            local userId = child:GetAttribute("UserId") or 0
            local bate = (filtroAtivo == "") or (nome:lower():find(filtroAtivo:lower(), 1, true) ~= nil)
            child.Visible = bate
            if bate then
                table.insert(btns, {
                    btn = child,
                    isFav = favoritos[userId] == true,
                    nome = nome,
                })
            end
        end
    end
    table.sort(btns, function(a, b)
        if a.isFav ~= b.isFav then return a.isFav end
        return a.nome:lower() < b.nome:lower()
    end)
    for i, item in ipairs(btns) do
        item.btn.LayoutOrder = i
    end
    return #btns
end

            playRandomTPSound()
            efeitoTP()

            task.delay(0.1, function()
                if portalEntrada and portalEntrada.Parent then
                    fecharPortal(portalEntrada)
                end
            end)

            task.delay(0.2, function()
                local targetChar = p.Character
                if not targetChar then
                    if tpLbl then tpLbl.Text = "▸ NO TARGET!" end
                    task.wait(1.5)
                    if tpLbl then tpLbl.Text = "▸ SELECT TARGET" end
                    return
                end

                local targetPivot = targetChar:GetPivot()
                local actualRoot = targetChar:FindFirstChild("HumanoidRootPart") or targetChar.PrimaryPart

                local currentChar = player.Character
                local currentRoot = currentChar and currentChar:FindFirstChild("HumanoidRootPart")

                if actualRoot then
                    local posAlvo = actualRoot.Position
                    local frenteAlvo = actualRoot.CFrame.LookVector
                    local posFinal = posAlvo + frenteAlvo * 10
                    currentRoot.CFrame = CFrame.new(posFinal, posAlvo)
                    if tpLbl then tpLbl.Text = "▸ SELECT TARGET" end
                else
                    currentRoot.CFrame = targetPivot * CFrame.new(0, 5, 0)
                    currentRoot.Anchored = true
                    pcall(function()
                        player:RequestStreamAroundAsync(targetPivot.Position)
                    end)

                    local timeout = os.clock() + 5
                    while os.clock() < timeout do
                        actualRoot = targetChar:FindFirstChild("HumanoidRootPart") or targetChar.PrimaryPart
                        if actualRoot then break end
                        task.wait(0.05)
                    end

                    if actualRoot then
                        local posAlvo = actualRoot.Position
                        local frenteAlvo = actualRoot.CFrame.LookVector
                        local posFinal = posAlvo + frenteAlvo * 10
                        currentRoot.CFrame = CFrame.new(posFinal, posAlvo)
                    else
                        if tpLbl then tpLbl.Text = "▸ GHOSTED!" end
                        task.wait(1.5)
                    end
                    currentRoot.Anchored = false
                    if tpLbl then tpLbl.Text = "▸ SELECT TARGET" end
                end

                task.wait(0.05)
                local cc2 = player.Character
                if cc2 then spawnPortalSaida(cc2) end
            end)
        end)
    end)

    return pBtn
end

local function refreshPlayerList()
    if refreshLock then return 0 end
    refreshLock = true

    local ok, resultado = pcall(function()
        for _, child in ipairs(scrollingList:GetChildren()) do
            if child:IsA("TextButton") then
                local uid = child:GetAttribute("UserId")
                if not uid or not playerBtns[uid] or playerBtns[uid] ~= child then
                    child:Destroy()
                end
            end
        end
        for userId, btn in pairs(playerBtns) do
            if not btn or not btn.Parent then
                playerBtns[userId] = nil
            end
        end
        for userId, btn in pairs(playerBtns) do
            local jogador = Players:GetPlayerByUserId(userId)
            if not jogador or jogador == player then
                if btn and btn.Parent then btn:Destroy() end
                playerBtns[userId] = nil
            end
        end
        local amizades = {}
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= player then
                local okF, isF = pcall(function() return player:IsFriendsWith(p.UserId) end)
                amizades[p.UserId] = okF and isF or false
            end
        end
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= player and not playerBtns[p.UserId] then
                playerBtns[p.UserId] = criarBotaoPlayer(p, 0, amizades[p.UserId] or false)
            end
        end
        local count = atualizarVisibilidade()
        scrollingList.CanvasSize = UDim2.new(0, 0, 0, count * 19)
        return count
    end)

    refreshLock = false
    if not ok then
        warn("[RickPrime] Erro no refreshPlayerList:", resultado)
        return 0
    end
    return resultado
end

local function contarPlayersVisiveis()
    local count = 0
    for _, child in ipairs(scrollingList:GetChildren()) do
        if child:IsA("TextButton") and child.Visible then
            count = count + 1
        end
    end
    return count
end

    if humanoid.Health <= 0 then atpUltPos = nil return end
    if atpCooldown then return end

    local agora = os.clock()
    if agora - atpUltimoCheck < ATP.Cooldown then return end

    if not atpUltPos then
        atpUltPos = hrp.Position
        atpUltimoCheck = agora
        return
    end

    if agora - atpUltimoRespawn < ATP.IgnorarJanela then
        atpUltPos = hrp.Position
        atpUltimoCheck = agora
        return
    end

    local distancia = (hrp.Position - atpUltPos).Magnitude

    if distancia > ATP.DistanciaMax then
        atpCooldown = true
        local rot = hrp.CFrame - hrp.CFrame.Position
        hrp.CFrame = CFrame.new(atpUltPos + Vector3.new(0, ATP.OffsetY, 0)) * rot
        hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        hrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
        alertaVermelho()
        efeitoTP()
        atpUltPos = atpUltPos + Vector3.new(0, ATP.OffsetY, 0)
        atpUltimoCheck = agora
        task.delay(0.3, function() atpCooldown = false end)
        return
    end

    atpUltPos = hrp.Position
    atpUltimoCheck = agora
end)

    atpUltPos = nil
    atpUltimoCheck = 0
    atpUltimoRespawn = os.clock()
end)

local function avBuscarSuperficie(posicao, character)
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = { character }
    params.IgnoreWater = false

    local candidatos = {}
    local raio = AV.RaioInicial

    while raio <= AV.RaioMaximoBusca do
        for i = 0, AV.PassosPorAnel - 1 do
            local angulo = (i / AV.PassosPorAnel) * math.pi * 2
            local x = posicao.X + math.cos(angulo) * raio
            local z = posicao.Z + math.sin(angulo) * raio
            local origem = Vector3.new(x, posicao.Y + AV.AlturaAcima, z)
            local direcao = Vector3.new(0, -(AV.AlturaAcima + AV.AlturaAbaixo), 0)
            local res = workspace:Raycast(origem, direcao, params)
            if res then
                local dot = res.Normal:Dot(Vector3.new(0, 1, 0))
                if dot >= AV.DotMinimoPisavel then
                    table.insert(candidatos, { posicao = res.Position, raio = raio })
                end
            end
        end
        if #candidatos > 0 and raio >= AV.RaioInicial * 4 then break end
        raio = raio + AV.RaioIncremento
    end

    if #candidatos == 0 then return nil end
    local melhor = candidatos[1]
    for _, c in ipairs(candidatos) do
        local dY = c.posicao.Y - melhor.posicao.Y
        if dY > AV.ToleranciaY then
            melhor = c
        elseif math.abs(dY) <= AV.ToleranciaY then
            if c.raio < melhor.raio then melhor = c end
        end
    end
    return melhor.posicao
end

local function avTeleportar(posicao, character)
    local hrp = character:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    local rot = hrp.CFrame - hrp.CFrame.Position

local function avTeleportarParaSuperficie(character)
    if avTpCooldown then return end

    genv.AntiTP_IgnorarProximo = true

    local sucesso, erro = pcall(function()
        local hrp = character:FindFirstChild("HumanoidRootPart")
        if not hrp then return end

        avEfeitoTeleporte()

        local superficie = avBuscarSuperficie(hrp.Position, character)
        if superficie then

        elseif avUltimaPosicaoSegura then

            end

        avTempoCaindo = 0

    task.delay(0.2, function()
        genv.AntiTP_IgnorarProximo = false
    end)

    local cooldown = avLocalSeguroAtivo and 0.3 or 1.5
    task.wait(cooldown)
    avTpCooldown = false
end

local function avRegistrarMorte()

    table.insert(avHistoricoMortes, agora)

    for i = #avHistoricoMortes, 1, -1 do
        if agora - avHistoricoMortes[i] > AV.JanelaMortes then
            table.remove(avHistoricoMortes, i)
        end
    end

    if #avHistoricoMortes >= AV.MortesLimite then
        avHistoricoMortes = {}
        task.spawn(function()
            task.wait(0.5)
            local char = player.Character
            if not char then return end
            local hrp = char:FindFirstChild("HumanoidRootPart")
            if not hrp then return end

            avEfeitoTeleporte()
            local superficie = avBuscarSuperficie(hrp.Position, char)
            if superficie then
                avTeleportar(superficie + Vector3.new(0, AV.OffsetY, 0), char)
            elseif avUltimaPosicaoSegura then
                avTeleportar(avUltimaPosicaoSegura + Vector3.new(0, AV.OffsetY, 0), char)
            else
                local spawn = workspace:FindFirstChildOfClass("SpawnLocation")
                if spawn then
                    avTeleportar(spawn.Position + Vector3.new(0, 5, 0), char)
                end
            end
            avLocalSeguroAtivo = true
            avLocalSeguroExpira = os.clock() + AV.LocalSeguroDuracao
        end)
    end
end

RunService.Heartbeat:Connect(function()
    if not antiVoidAtivo then return end
    local character = player.Character
    if not character then
        avTempoCaindo = 0
        avUltimaPos = nil
        return
    end

    local hrp = character:FindFirstChild("HumanoidRootPart")
    local humanoid = character:FindFirstChildOfClass("Humanoid")

    if not hrp or not humanoid then
        avTempoCaindo = 0
        avUltimaPos = nil
        return
    end
    if humanoid.Health <= 0 then
        avTempoCaindo = 0
        avUltimaPos = nil
        return
    end

    if avDesativado and os.clock() < avDesativadoExpira then
        avTempoCaindo = 0
        avUltimaPos = nil
        return
    end

    if hrp.Position.Y > 50 then
        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        params.FilterDescendantsInstances = { character }
        local origem = hrp.Position + Vector3.new(0, 2, 0)
        local direcao = Vector3.new(0, -15, 0)
        if workspace:Raycast(origem, direcao, params) then
            avUltimaPosicaoSegura = hrp.Position
        end
    end

    local agora = os.clock()
    local dt = agora - avUltimoTempo
    avUltimoTempo = agora

    local velocidadeY = hrp.AssemblyLinearVelocity.Y
    local caindoPorVel = velocidadeY < AV.VelocidadeMinimaCaindo
    local state = humanoid:GetState()
    local caindoPorEstado = (state == Enum.HumanoidStateType.Freefall)
        or (state == Enum.HumanoidStateType.FallingDown)

    local caindoPorPos = false
    if avUltimaPos then
        local deltaY = hrp.Position.Y - avUltimaPos.Y
        caindoPorPos = (deltaY < -0.5) and (velocidadeY < -5)
    end
    avUltimaPos = hrp.Position

    if velocidadeY > 5 then
        avTempoCaindo = 0
        return
    end

    local estaCaindo = caindoPorVel or caindoPorEstado or caindoPorPos
    local limiteAtual = (avLocalSeguroAtivo and os.clock() < avLocalSeguroExpira)
        and AV.TempoCaindoLimite or AV.TempoCaindoNormal

    if estaCaindo then
        avTempoCaindo = avTempoCaindo + dt
        if avTempoCaindo >= limiteAtual and not avTpCooldown then
            avTeleportarParaSuperficie(character)
        end
    else
        avTempoCaindo = 0
    end
end)
